cmake_minimum_required(VERSION 3.24)
project(mode_indicator_regression LANGUAGES CXX)
# Re-extract whenever a production method changes; never test stale copies.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/../../src/text_service.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/../../src/inline_preedit.cpp")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/text_service.cpp" SOURCE)
foreach(method MaybeShowModeIndicator FlushModeIndicator LoadApplicationSettings CaptureCaretRect)
  string(FIND "${SOURCE}" "void TextService::${method}(" start)
  string(SUBSTRING "${SOURCE}" ${start} -1 remaining)
  string(FIND "${remaining}" "\n}\n" end)
  math(EXPR length "${end}+3")
  string(SUBSTRING "${remaining}" 0 ${length} body)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${method}.inc" "${body}")
endforeach()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/inline_preedit.cpp" INLINE_SOURCE)
string(FIND "${INLINE_SOURCE}" "bool InlinePreedit::LastCaret(" start)
string(SUBSTRING "${INLINE_SOURCE}" ${start} -1 remaining)
string(FIND "${remaining}" "\n}\n" end)
math(EXPR length "${end}+3")
string(SUBSTRING "${remaining}" 0 ${length} body)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/LastCaret.inc" "${body}")
add_executable(mode-indicator-regression regression.cpp)
target_include_directories(mode-indicator-regression PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(mode-indicator-regression PRIVATE cxx_std_20)
target_compile_options(mode-indicator-regression PRIVATE /utf-8)
target_link_libraries(mode-indicator-regression PRIVATE user32 advapi32 uuid)

enable_testing()
add_test(NAME mode-indicator-regression COMMAND mode-indicator-regression)
