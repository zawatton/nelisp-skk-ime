cmake_minimum_required(VERSION 3.24)
project(nelisp_ime_keymap LANGUAGES CXX)
enable_testing()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/input-rules.conf" INPUT_RULES_TEMPLATE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/input_rules_template.inc" "R\"rules(${INPUT_RULES_TEMPLATE})rules\"")
add_library(keymap-core core.cpp config.cpp rules.cpp)
target_include_directories(keymap-core PUBLIC "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(keymap-core PUBLIC cxx_std_20)
add_executable(nelisp-ime-keymap runner.cpp editor.cpp rules_editor.cpp)
add_executable(keymap-tests tests.cpp)
add_executable(keymap-layout-test layout-test.cpp)
foreach(t keymap-core nelisp-ime-keymap keymap-tests keymap-layout-test)
 target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8)
 target_compile_definitions(${t} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
endforeach()
target_link_libraries(nelisp-ime-keymap PRIVATE keymap-core user32 advapi32 imm32 wtsapi32 shell32 ole32 comctl32)
target_link_libraries(keymap-tests PRIVATE keymap-core)
target_link_libraries(keymap-layout-test PRIVATE keymap-core)
add_test(NAME native-layout-tables COMMAND keymap-layout-test)
add_test(NAME core COMMAND keymap-tests)
add_test(NAME benchmark COMMAND keymap-tests --benchmark)
add_test(NAME hook-self-test COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File ${CMAKE_CURRENT_SOURCE_DIR}/test-self.ps1 -Exe $<TARGET_FILE:nelisp-ime-keymap>)
add_test(NAME editor-self-test COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File ${CMAKE_CURRENT_SOURCE_DIR}/test-self.ps1 -Exe $<TARGET_FILE:nelisp-ime-keymap> -Editor)
add_test(NAME rules-editor-self-test COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File ${CMAKE_CURRENT_SOURCE_DIR}/test-rules-self.ps1 -Exe $<TARGET_FILE:nelisp-ime-keymap>)
add_test(NAME private-lifecycle COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File ${CMAKE_CURRENT_SOURCE_DIR}/test-lifecycle.ps1 -Exe $<TARGET_FILE:nelisp-ime-keymap>)

add_test(NAME private-install COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File ${CMAKE_CURRENT_SOURCE_DIR}/test-install.ps1 -Exe $<TARGET_FILE:nelisp-ime-keymap>)

set_tests_properties(native-layout-tables core benchmark hook-self-test editor-self-test PROPERTIES TIMEOUT 30)
set_tests_properties(private-lifecycle private-install PROPERTIES TIMEOUT 90)
