cmake_minimum_required(VERSION 3.24)
project(local_first_dictionary_prototype LANGUAGES CXX)
add_library(local-first-index STATIC mmap_index.cpp)
target_compile_features(local-first-index PUBLIC cxx_std_20)
target_compile_definitions(local-first-index PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(local-first-index PRIVATE /W4 /permissive- /utf-8)
add_executable(local-first-bench bench.cpp)
target_link_libraries(local-first-bench PRIVATE local-first-index psapi)
target_compile_options(local-first-bench PRIVATE /W4 /permissive- /utf-8)
enable_testing()
add_test(NAME local-first-index-selftest COMMAND local-first-bench selftest "${CMAKE_CURRENT_BINARY_DIR}/selftest")
